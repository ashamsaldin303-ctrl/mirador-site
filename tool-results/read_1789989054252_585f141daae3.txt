   131→   131→"use client";

   133→   133→import Lenis from "lenis";
   134→   134→import gsap from "gsap";
   135→   135→import { ScrollTrigger } from "gsap/ScrollTrigger";
   136→   136→
   137→   137→gsap.registerPlugin(ScrollTrigger);
   138→   138→
   139→   139→export function SmoothScroll({ children }: { children: React.ReactNode }) {
   140→   140→  useEffect(() => {
   141→   141→    if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
   142→   142→    const lenis = new Lenis({ duration: 1.15, wheelMultiplier: 0.9 });
   143→   143→    lenis.on("scroll", ScrollTrigger.update);
   144→   144→    const raf = (t: number) => lenis.raf(t * 1000);
   145→   145→    gsap.ticker.add(raf);
   146→   146→    gsap.ticker.lagSmoothing(0);
   147→   147→    return () => { gsap.ticker.remove(raf); lenis.destroy(); };
   148→   148→  }, []);
   149→   149→  return <>{children}</>;
   150→   150→}
   151→   151→```
